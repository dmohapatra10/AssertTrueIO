class QAAgent:
    def __init__(self, objective, instructions, tools, client, model):
        self.objective = objective
        self.instructions = instructions  # Chapter 2's own
                                             # careful instruction-
                                             # writing principles
        self.tools = tools                # Chapter 2's own
                                             # tool-definition format
        self.client = client
        self.model = model
        self.max_steps = 15               # Chapter 2's own loop
                                             # guardrail

    def run(self, page):
        history = []
        for step in range(self.max_steps):
            prompt = self._build_prompt(history, page)
            response = self.client.messages.create(
                model=self.model, max_tokens=1024,
                system=self.instructions, tools=self.tools,
                messages=[{"role": "user", "content": prompt}]
            )
            if self._is_task_complete(response):
                return self._build_report(history, success=True)
            tool_call = self._extract_tool_call(response)
            result = execute_tool(tool_call.name, tool_call.input, page)
            history.append({"step": step, "action": tool_call, "result": result})
        return self._build_report(history, success=False)
