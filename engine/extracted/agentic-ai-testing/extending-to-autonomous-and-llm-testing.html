def verify_layer3_independence(codebase_modules):
    """Statically checks that Chapter 10, 11, and 12's
    implementation modules never import from or call into
    each other, catching accidental coupling introduced
    during future development."""
    layer3_modules = {
        "chapter10_api_coverage": codebase_modules["chapter10_api_coverage"],
        "chapter11_llm_evaluation": codebase_modules["chapter11_llm_evaluation"],
        "chapter12_agent_testing": codebase_modules["chapter12_agent_testing"],
    }
    violations = []
    for module_name, module_source in layer3_modules.items():
        for other_name in layer3_modules:
            if other_name != module_name and other_name in module_source:
                violations.append(
                    f"{module_name} appears to reference {other_name} — "
                    f"genuine Layer 3 independence may have been broken.")
    return violations if violations else "Layer 3 independence confirmed."
